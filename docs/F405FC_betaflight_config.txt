# Betaflight / STM32F405 (S405) 4.5.x  --  unified-target style CLI config
# Board: F405FC (custom), STM32F405RGT6 LQFP-64
# Companion document: design/pinmap.md (pin table + DMA proof). Rev C, 2026-10-11
# Rev C: analog OSD (AT7456E) removed -> SPI2 unused, M7 (B07) gets DMA1 S3 (DShot);
#        PINIO1 (C15) = VTX OFF when high (pulls 10 V buck EN low).
# (Gate 0 audit fixes: beeper->PB2, CAM_SEL->PC13, ESC TLM->UART4 RX PA1,
#  M7 PB7 / M8 PA0 PWM-servo only, VBUS detect removed, UART5 full spare).
#
# Apply on a STM32F405 build with CLI 'defaults nosave' then paste. With a Configurator cloud build, tick the drivers
# explicitly: ICM42688P, BMI270, BMP388 (covers BMP390), the IIS2MDC/LIS2MDL mag driver and W25N01G flash; they are not
# all in a default build. The config.h appendix at the bottom is a sketch for a custom build: its macro names were not all
# found in the Betaflight sources on disk (Gate 6 m-6) and it has not been compiled.
# baro_hardware / mag_hardware stay AUTO: the exact Betaflight enum names for BMP390 / IIS2MDC were not verified (Gate 6 m-7).

timer B00 AF2
# pin B00: TIM3 CH3 (AF2)          MOTOR 1
timer B01 AF2

timer B10 AF1
# pin B10: TIM2 CH3 (AF1)          MOTOR 3
timer B11 AF1
# pin B11: TIM2 CH4 (AF1)          MOTOR 4
timer C08 AF3
# pin C08: TIM8 CH3 (AF3)          MOTOR 5 (extra pad)
timer C09 AF3
# pin C09: TIM8 CH4 (AF3)          MOTOR 6 (extra pad)
timer A08 AF1
# pin A08: TIM1 CH1 (AF1)          LED_STRIP
timer B07 AF2
# pin B07: TIM4 CH2 (AF2)          MOTOR 7 (extra pad, DShot capable)
timer A00 AF2
# pin A00: TIM5 CH1 (AF2)          MOTOR 8 (extra pad, PWM/servo only)

# ---------------------------------------------------------------- dma
# Peripherals (all pinned explicitly; see pinmap.md section 3)
dma SPI_SDO 1 0
# SPI_SDO 1: DMA2 Stream 3 Channel 3
dma SPI_SDI 1 0
# SPI_SDI 1: DMA2 Stream 0 Channel 3
dma SPI_SDO 3 0
# SPI_SDO 3: DMA1 Stream 5 Channel 0
dma SPI_SDI 3 0
# SPI_SDI 3: DMA1 Stream 0 Channel 0
dma ADC 1 1
# ADC 1: DMA2 Stream 4 Channel 0   (opt 0 = DMA2 S0 would steal SPI1 RX)

# Timer pins
dma pin B00 0
# pin B00: DMA1 Stream 7 Channel 5
dma pin B01 0
# pin B01: DMA1 Stream 2 Channel 5
dma pin B10 0
# pin B10: DMA1 Stream 1 Channel 3
dma pin B11 1
# pin B11: DMA1 Stream 6 Channel 3 (opt 0 = DMA1 S7 is used by B00)
dma pin C08 0
# pin C08: DMA2 Stream 2 Channel 0
dma pin C09 0
# pin C09: DMA2 Stream 7 Channel 7
dma pin A08 0
# pin A08: DMA2 Stream 6 Channel 0
dma pin B07 0
# pin B07: DMA1 Stream 3 Channel 2 (free since SPI2/OSD removed) -> DShot capable
dma pin A00 NONE
# pin A00: no DMA (only option DMA1 S2 Ch6 = M2)     -> PWM/servo only, no DShot

# DMA1: S0 SPI3_RX | S1 M3 | S2 M2 | S3 M7 | S4 free | S5 SPI3_TX | S6 M4 | S7 M1
# DMA2: S0 SPI1_RX | S1 free | S2 M5 | S3 SPI1_TX | S4 ADC1 | S5 free | S6 LED_STRIP | S7 M6
# M8 (A00): no DMA. Use as a servo (resource SERVO n A00) or PWM/Oneshot.
# An 8-motor mixer with DShot will NOT work on M8 (7 DShot motors do).

# ---------------------------------------------------------------- features
feature RX_SERIAL
feature OSD
feature LED_STRIP
feature TELEMETRY
feature ESC_SENSOR

# ---------------------------------------------------------------- serial
# serial <id> <functionMask> <mspBaud> <gpsBaud> <telemetryBaud> <blackboxBaud>
# UART1 (0): HD VTX MSP DisplayPort  -> MSP(1) + VTX_MSP(131072)
serial 0 131073 115200 57600 0 115200
# UART2 (1): ELRS receiver           -> SERIAL_RX(64)
serial 1 64 115200 57600 0 115200
# UART3 (2): GPS                     -> GPS(2)
serial 2 2 115200 57600 0 115200
# UART4 (3): ESC telemetry, RX-only on PA1 (TX pin PA0 is MOTOR 8) -> ESC_SENSOR(1024)
serial 3 1024 115200 57600 0 115200
# UART5 (4): spare, full TX+RX (PD2 optionally gets HD SBUS/R2 via DNP 0R, no inverter)
serial 4 0 115200 57600 0 115200
# UART6 (5): spare
serial 5 0 115200 57600 0 115200
# VCP (20): USB MSP
serial 20 1 115200 57600 0 115200

# ---------------------------------------------------------------- master settings
set system_hse_mhz = 8

# Gyros: both on SPI1, separate CS + EXTI
set gyro_1_bustype = SPI
set gyro_1_spibus = 1
set gyro_1_sensor_align = CW270FLIP
set gyro_2_bustype = SPI
set gyro_2_spibus = 1
set gyro_2_sensor_align = CW0FLIP
set gyro_to_use = FIRST
# gyro_1 = ICM-42688-P (CS A04 / EXTI C04), gyro_2 = BMI270 (CS A15 / EXTI C14)
# Alignment (PROVISIONAL, Gate 3 G3-13): both IMUs on the bottom side, rot 0; board front = J3 (VTX) edge, marked "FRONT" on silk.
# Derived from datasheet axis figures, not yet bench-verified: confirm with the tilt test in docs/assembly_and_bringup.md §4.

# Barometer BMP390 (BF BMP388 driver) on I2C1, SDO strapped to GND -> 0x76
set baro_bustype = I2C
set baro_i2c_device = 1
set baro_hardware = AUTO

# Magnetometer IIS2MDC (BF LIS2MDL driver, fixed 0x1E) on I2C1 (shared with GPS pads)
set mag_bustype = I2C
set mag_i2c_device = 1
set mag_hardware = AUTO

# I2C1: on-board 2.2k pull-ups fitted; 400 kHz for the off-board GPS-mag cable
set i2c1_pullup = OFF
set i2c1_clockspeed_khz = 400

# Blackbox W25N01GV on SPI3
set flash_spi_bus = 3
set blackbox_device = SPIFLASH

# Motors: timer DShot (NOT bit-bang: bit-bang needs TIM1/TIM8 and disables SPI DMA on DMA2)
set motor_pwm_protocol = DSHOT600
set dshot_bidir = ON
set dshot_bitbang = OFF
set dshot_burst = OFF

# Battery / current (ADC1: VBAT C00, CURR C01, RSSI C05)
set battery_meter = ADC
set current_meter = ADC
set vbat_scale = 110
# ibata_scale/offset depend on the ESC's current sensor -- placeholder, calibrate per ESC
set ibata_scale = 400
set ibata_offset = 0

# Beeper: PB2 (BOOT1 strap, 10k pulldown) drives an N-MOSFET gate, active HIGH.
# BF drives the pin LOW for "beep" unless inverted -> inversion ON = HIGH for beep.
set beeper_inversion = ON
set beeper_od = OFF

# Status LEDs PB14/PB15 wired 3V3 -> R -> LED -> pin (sink, low = on) = BF non-inverted
set led_inversion = 0

# PINIO1 (C15) = VTX power cut: HIGH turns Q2 on, which pulls the 10 V buck EN low (VTX OFF).
#   1 = output, normal: USER1 off -> pin LOW -> VTX ON (also ON through reset/DFU via 100k gate pulldown)
#   USER1 on -> pin HIGH -> VTX OFF (pit/power cut)
# PINIO2 (C13) = camera select (low = CAM1). 1 = output, normal: USER2 on -> CAM2
set pinio_config = 1,1,1,1
set pinio_box = 40,41,255,255

# Receiver
set serialrx_provider = CRSF

# OSD: no analog OSD chip on this board (D-01). MSP DisplayPort over UART1 to the HD VTX.
set osd_displayport_device = MSP
